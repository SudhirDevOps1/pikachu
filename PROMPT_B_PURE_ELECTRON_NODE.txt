2026 ANALYSIS: PURE NODE.JS — HONEST TRADEOFFS
--------------------------------------------------------------------------------

WHY CHOOSE THIS PATH:
  + Single process, no spawn overhead (cold start <1s vs ~2s sidecar)
  + Smallest bundle possible — no PyInstaller exe inside Electron
  + No Python PATH issues, no PyInstaller antivirus flags
  + Simpler CI/CD — no Python in the build pipeline
  + One language for the whole team (JS/TS developers)
  + Tauri-compatible thinking (easier migration to Tauri later if needed)
  + node-gyp native modules compile faster than PyInstaller freezing
  + Web Speech API + whisper.js give zero-install voice recognition in 2026

WHY THIS IS HARDER THAN IT SOUNDS:
  + Vosk Node npm is NOT as mature as the Python version (fewer examples)
  + @nut-tree/nut-js requires node-gyp + Visual C++ Build Tools on Windows
  + Native modules must be rebuilt for each Electron version (electron-rebuild)
  + keyboard/mouse macro recording is less reliable cross-platform in Node
  + Background wake word detection is difficult when renderer is hidden/throttled
  + OCR with tesseract.js (WASM) is 3-5x slower than pytesseract
  + pyttsx3 offline TTS has no direct Node equivalent of equal quality

'use strict';
const si = require("systeminformation");
const os = require("os");

async function getCpu() {
  const load = await si.currentLoad();
  const cpu  = await si.cpu();
  return {
    success: true,
    message: `CPU ${Math.round(load.currentLoad)}%`,
    data: {
      percent: Math.round(load.currentLoad),
      cores: os.cpus().length,
      model: cpu.brand,
      speed: cpu.speed,
    },
  };
}

async function getRam() {
  const mem = await si.mem();
  return {
    success: true,
    message: `RAM ${Math.round((mem.used / mem.total) * 100)}%`,
    data: {
      percent: Math.round((mem.used / mem.total) * 100),
      used_gb: +(mem.used / 2**30).toFixed(1),
      total_gb: +(mem.total / 2**30).toFixed(1),
    },
  };
}

async function getBattery() {
  const b = await si.battery();
  return {
    success: true,
    message: `बैटरी ${b.percent}%`,
    data: { percent: b.percent, plugged: b.isCharging, hasBattery: b.hasBattery },
  };
}

async function getDisk() {
  const [fs] = await si.fsSize();
  if (!fs) return { success: false, message: "डिस्क नहीं मिली", data: null };
  return {
    success: true,
    message: `डिस्क ${Math.round((fs.used/fs.size)*100)}%`,
    data: {
      percent: Math.round((fs.used / fs.size) * 100),
      free_gb: +((fs.size - fs.used) / 2**30).toFixed(1),
      total_gb: +(fs.size / 2**30).toFixed(1),
    },
  };
}

async function getFullReport() {
  const [cpu, ram, battery, disk] = await Promise.all([getCpu(), getRam(), getBattery(), getDisk()]);
  const nets = require("os").networkInterfaces();
  let ip = "127.0.0.1";
  for (const iface of Object.values(nets)) {
    for (const info of (iface || [])) {
      if (info.family === "IPv4" && !info.internal) { ip = info.address; break; }
    }
  }
  return {
    success: true,
    message: "सिस्टम रिपोर्ट",
    data: { ...cpu.data, ...ram.data, ...battery.data, ...disk.data, ip, hostname: os.hostname() },
  };
}

module.exports = { getCpu, getRam, getBattery, getDisk, getFullReport };

'use strict';
const fs = require("fs/promises");
const path = require("path");
const os = require("os");

const BLOCKED = [
  /^[a-zA-Z]:\\Windows/i, /^[a-zA-Z]:\\Program Files/i,
  /^\/System/i, /^\/usr/i, /^\/etc/i, /^\/bin/i, /^\/sbin/i
];

function resolvePath(p) {
  if (!p) return path.join(os.homedir(), "Desktop");
  const low = p.toLowerCase().trim();
  const map = { desktop:"Desktop", documents:"Documents", downloads:"Downloads",
                pictures:"Pictures", music:"Music", videos:"Videos" };
  for (const [k, v] of Object.entries(map)) {
    if (low === k || low.startsWith(k + "/") || low.startsWith(k + "\\")) {
      const rest = p.slice(k.length).replace(/^[\\/]/, "");
      return rest ? path.join(os.homedir(), v, rest) : path.join(os.homedir(), v);
    }
  }
  const resolved = path.resolve(p);
  return path.isAbsolute(p) ? resolved : path.join(os.homedir(), p);
}

function isSafe(p) {
  return !BLOCKED.some((re) => re.test(p));
}

const ok  = (msg, data = null) => ({ success: true, message: msg, data });
const err = (msg)              => ({ success: false, message: msg, data: null });

module.exports = {
  async createFile(p, content = "") {
    const full = resolvePath(p);
    if (!isSafe(full)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    await fs.mkdir(path.dirname(full), { recursive: true });
    await fs.writeFile(full, content, "utf8");
    return ok(`फाइल बनी: ${path.basename(full)}`);
  },
  async createFolder(p) {
    const full = resolvePath(p);
    if (!isSafe(full)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    await fs.mkdir(full, { recursive: true });
    return ok(`फोल्डर बना: ${path.basename(full)}`);
  },
  async readFile(p) {
    const full = resolvePath(p);
    if (!isSafe(full)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    const content = await fs.readFile(full, "utf8");
    return ok(`पढ़ा: ${path.basename(full)}`, { content: content.slice(0, 20000) });
  },
  async writeFile(p, content = "") {
    const full = resolvePath(p);
    if (!isSafe(full)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    await fs.writeFile(full, content, "utf8");
    return ok(`सेव: ${path.basename(full)}`);
  },
  async deleteFile(p) {
    const full = resolvePath(p);
    if (!isSafe(full)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    try { await fs.rm(full, { recursive: true, force: true }); }
    catch { return err("फाइल नहीं मिली या access denied।"); }
    return ok(`डिलीट: ${path.basename(full)}`);
  },
  async renameFile(oldPath, newPath) {
    const src = resolvePath(oldPath);
    const dst = resolvePath(newPath);
    if (!isSafe(src) || !isSafe(dst)) return err("सुरक्षा: पथ प्रतिबंधित है।");
    await fs.rename(src, dst);
    return ok(`रीनेम: ${path.basename(src)} → ${path.basename(dst)}`);
  },
  async listDirectory(p) {
    const full = resolvePath(p);
    const entries = await fs.readdir(full, { withFileTypes: true });
    const items = entries.slice(0, 50).map(e => ({ name: e.name, is_dir: e.isDirectory() }));
    return ok(`${items.length} आइटम`, { path: full, items });
  },
};

  async listDrives() {
    const drives = await drivelist.list();
    const result = drives
      .filter(d => d.mountpoints.length > 0)
      .map(d => ({
        name: d.mountpoints[0].path,
        description: d.description,
        size: d.size || 0,
        isReadOnly: d.isReadOnly,
      }));

    // Get usage via systeminformation
    const si = require("systeminformation");
    const fsData = await si.fsSize();
    const enriched = result.map(d => {
      const fs = fsData.find(f => f.mount === d.name || f.mount === d.name.replace("\\", ""));
      return fs
        ? { ...d, total: fs.size, free: fs.size - fs.used, used: fs.used,
            percent: Math.round((fs.used / fs.size) * 100) }
        : { ...d, total: d.size, free: 0, used: 0, percent: 0 };
    });

    return { success: true, message: `${enriched.length} ड्राइव मिले।`, data: { drives: enriched } };
  },

module.exports = {
  async listProcesses() {
    const procs = await psList();
    const sorted = procs
      .sort((a, b) => (b.memory || 0) - (a.memory || 0))
      .slice(0, 30)
      .map(p => ({ pid: p.pid, name: p.name, cpu: 0, ram: 0 }));
    return { success: true, message: "प्रोसेस सूची", data: { items: sorted } };
  },
  async killProcess(nameOrPid) {
    try {
      if (IS_WIN) {
        const flag = /^\d+$/.test(nameOrPid) ? `/PID ${nameOrPid}` : `/IM ${nameOrPid}.exe`;
        await execAsync(`taskkill /F ${flag}`);
      } else {
        await execAsync(`pkill -f "${nameOrPid}"`);
      }
      return { success: true, message: `${nameOrPid} बंद कर दिया।`, data: null };
    } catch {
      return { success: false, message: "प्रोसेस नहीं मिला।", data: null };
    }
  },
};

'use strict';

const PROVIDERS = {
  groq:       { url: "https://api.groq.com/openai/v1/chat/completions",    env: "GROQ_API_KEY",       model: "llama-3.3-70b-versatile" },
  cerebras:   { url: "https://api.cerebras.ai/v1/chat/completions",        env: "CEREBRAS_API_KEY",   model: "llama-3.3-70b" },
  mistral:    { url: "https://api.mistral.ai/v1/chat/completions",         env: "MISTRAL_API_KEY",    model: "mistral-small-latest" },
  deepseek:   { url: "https://api.deepseek.com/chat/completions",          env: "DEEPSEEK_API_KEY",   model: "deepseek-chat" },
  openrouter: { url: "https://openrouter.ai/api/v1/chat/completions",      env: "OPENROUTER_API_KEY", model: "meta-llama/llama-3.3-70b-instruct:free" },
};

const ORDER = ["groq", "cerebras", "mistral", "deepseek", "openrouter"];
const SYSTEM = "You are Pika, a friendly personal AI assistant on the user's PC. Default language Hindi.";
const history = [];
let currentProvider = ORDER.find(p => process.env[PROVIDERS[p].env]) || "groq";

module.exports = {
  setProvider(name) { if (PROVIDERS[name]) { currentProvider = name; return true; } return false; },
  getProvider() { return currentProvider; },
  available() { return ORDER.filter(p => process.env[PROVIDERS[p].env]); },

  async* stream(text) {
    history.push({ role: "user", content: text });
    if (history.length > 40) history.splice(0, history.length - 40);

    const providers = [currentProvider, ...ORDER.filter(p => p !== currentProvider)].filter(p => process.env[PROVIDERS[p].env]);
    if (!providers.length) {
      yield { chunk: "Settings mein koi API key set nahi hai। Demo mode chal raha hai।", provider: "local_fallback", done: false };
      yield { chunk: "", provider: "local_fallback", done: true };
      return;
    }

    for (const provider of providers) {
      const { url, env, model } = PROVIDERS[provider];
      const messages = [{ role: "system", content: SYSTEM }, ...history.slice(-20)];
      let full = "";
      let failed = false;

      try {
        const res = await fetch(url, {
          method: "POST",
          headers: { "Authorization": `Bearer ${process.env[env]}`, "Content-Type": "application/json" },
          body: JSON.stringify({ model, messages, stream: true, temperature: 0.7, max_tokens: 2048 }),
          signal: AbortSignal.timeout(60000),
        });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);

        const reader = res.body.getReader();
        const decoder = new TextDecoder();
        let buf = "";
        while (true) {
          const { done, value } = await reader.read();
          if (done) break;
          buf += decoder.decode(value, { stream: true });
          const lines = buf.split("\n");
          buf = lines.pop();
          for (const line of lines) {
            const t = line.replace(/^data: /, "").trim();
            if (!t || t === "[DONE]") continue;
            try {
              const delta = JSON.parse(t).choices[0].delta.content || "";
              if (delta) { full += delta; yield { chunk: delta, provider, done: false }; }
            } catch {}
          }
        }
      } catch (e) {
        console.error(`[llm] ${provider} failed:`, e.message);
        failed = true;
      }

      if (!failed && full) {
        history.push({ role: "assistant", content: full });
        yield { chunk: "", provider, done: true };
        return;
      }
    }

    yield { chunk: "सभी AI providers अभी उपलब्ध नहीं हैं। बाद में फिर पूछें।", provider: "local_fallback", done: false };
    yield { chunk: "", provider: "local_fallback", done: true };
  },
};

      } catch { /* fallback */ }
    }

    return new Promise((resolve) => {
      say.speak(text, null, 1.0, (err) => {
        if (err) resolve({ success: false, message: err.message });
        else resolve({ success: true, audio: null, format: "native" });
      });
    });
  },

function initSchema(db) {
  db.exec(`
    CREATE TABLE IF NOT EXISTS settings (
      key TEXT PRIMARY KEY, value TEXT NOT NULL,

    );
    CREATE TABLE IF NOT EXISTS reminders (
      id TEXT PRIMARY KEY, text TEXT NOT NULL,

      status TEXT NOT NULL DEFAULT 'active'
    );
    CREATE INDEX IF NOT EXISTS idx_rem ON reminders(status, trigger_time);
    CREATE TABLE IF NOT EXISTS command_log (
      id TEXT PRIMARY KEY, command_type TEXT, category TEXT, action TEXT,
      params TEXT, status TEXT, message TEXT, duration_ms INTEGER DEFAULT 0,
      timestamp TEXT NOT NULL DEFAULT (datetime('now'))
    );
    CREATE INDEX IF NOT EXISTS idx_cmd ON command_log(timestamp DESC);
    CREATE TABLE IF NOT EXISTS snippets (
      trigger TEXT PRIMARY KEY, content TEXT NOT NULL,
      use_count INTEGER DEFAULT 0, created_at TEXT NOT NULL DEFAULT (datetime('now'))
    );
    CREATE TABLE IF NOT EXISTS sessions (
      id TEXT PRIMARY KEY, started_at TEXT NOT NULL DEFAULT (datetime('now')),
      ended_at TEXT, message_count INTEGER DEFAULT 0
    );
    CREATE TABLE IF NOT EXISTS conversation (
      id TEXT PRIMARY KEY, role TEXT NOT NULL, content TEXT NOT NULL,
      provider TEXT, session_id TEXT NOT NULL,
      timestamp TEXT NOT NULL DEFAULT (datetime('now')),
      FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE
    );
  `);

module.exports = {
  // Settings
  setSetting(key, value) {
    getDb().prepare(
      "INSERT INTO settings(key, value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value, updated_at=datetime('now')"
    ).run(key, typeof value === "string" ? value : JSON.stringify(value));
  },
  getSetting(key, def = null) {
    const row = getDb().prepare("SELECT value FROM settings WHERE key=?").get(key);
    return row ? row.value : def;
  },
  // Reminders

    const id = randomUUID();

  },

    const cutoff = new Date(Date.now() - days * 86400000).toISOString();
    let total = 0;
    for (const [t, c] of [["command_log","timestamp"],["conversation","timestamp"]]) {
      const r = getDb().prepare(`DELETE FROM ${t} WHERE ${c} < ?`).run(cutoff);
      total += r.changes;

    getDb().exec("VACUUM");

function safeEval(expr) {
  // Whitelist-only math using mathjs limited scope
  try {
    const { create, all } = require("mathjs");
    const math = create(all, {});
    const scope = {};
    const result = math.evaluate(expr, scope);
    if (typeof result !== "number" || !isFinite(result)) throw new Error("not a number");
    if (Math.abs(result) > 1e15) throw new Error("too large");
    return { success: true, message: `${expr} = ${result}`, data: { result } };
  } catch (e) {
    return { success: false, message: "अमान्य एक्सप्रेशन।", data: null };
  }
}

async function route(msg, emitEvent) {
  const { type, category, action, params = {} } = msg;
  const p = params || {};

  if (type === "command" && category === "_confirm") {
    const cid = p.confirmation_id;
    if (action === "approve" && pendingConfirm.has(cid)) {
      const original = pendingConfirm.get(cid);
      pendingConfirm.delete(cid);
      return route(original, emitEvent);
    }
    pendingConfirm.delete(cid);
    return { success: true, message: "रद्द किया गया।", data: null };
  }

  const key = `${category}:${action}`;
  if (CONFIRM_REQUIRED.has(key) && !p.confirmed) {
    const cid = randomUUID();
    pendingConfirm.set(cid, { ...msg, params: { ...p, confirmed: true } });
    return { type: "response", status: "confirmation_required",
             message: `क्या आप ${category}/${action} करना चाहते हैं?`,
             confirmation_id: cid, data: null };
  }

  // Route to services
  try {
    switch (category) {
      case "system": return await system[{
        cpu:"getCpu", ram:"getRam", battery:"getBattery", disk:"getDisk",
        full_report:"getFullReport", ip:"getNetworkInfo",
      }[action] || "getCpu"]?.() ?? { success:false, message:`Unknown info: ${action}` };

      case "power":
      case "system":
        if (["shutdown","restart","sleep","lock","hibernate","logoff"].includes(action))
          return await power[action](p.delay);
        break;

      case "info":
        if (action === "ip") return await network.getNetworkInfo();
        if (action === "cpu") return await system.getCpu();
        if (action === "ram") return await system.getRam();
        if (action === "battery") return await system.getBattery();
        if (action === "disk") return await system.getDisk();
        if (action === "full_report") return await system.getFullReport();
        if (action === "time") return { success:true, message: new Date().toLocaleTimeString("hi-IN"), data:null };
        if (action === "date") return { success:true, message: new Date().toLocaleDateString("hi-IN", { weekday:"long", day:"numeric", month:"long", year:"numeric" }), data:null };
        break;

      case "volume":
        if (action === "up")      return await volume.volumeUp(p.amount);
        if (action === "down")    return await volume.volumeDown(p.amount);
        if (action === "set")     return await volume.volumeSet(p.percent || p.level || 50);
        if (action === "mute" || action === "unmute") return await volume.mute();
        break;

      case "files":
        if (action === "create_file")   return await file.createFile(p.path, p.content);
        if (action === "create_folder") return await file.createFolder(p.path);
        if (action === "read")          return await file.readFile(p.path);
        if (action === "write")         return await file.writeFile(p.path, p.content);
        if (action === "delete")        return await file.deleteFile(p.path);
        if (action === "rename")        return await file.renameFile(p.path, p.new_path);
        if (action === "list")          return await file.listDirectory(p.path);
        if (action === "open_explorer") { shell.openPath(p.path || os.homedir()); return { success:true, message:"खोल दिया।", data:null }; }
        break;

      case "disk":
        if (action === "list_drives") return await drive.listDrives();
        break;

      case "screen":
        if (action === "screenshot") return await screen.takeScreenshot();
        break;

      case "processes":
        if (action === "list") return await proc.listProcesses();
        if (action === "kill") return await proc.killProcess(p.name_or_pid);
        break;

      case "network":
        if (action === "ip") return await network.getNetworkInfo();
        break;

      case "apps":
        if (action === "open") {
          const name = (p.name || "").toLowerCase().trim();
          const exe = APP_MAP[name];
          const url = URL_MAP[name];
          if (url) { shell.openExternal(url); return { success:true, message:`${p.name} खोला।`, data:null }; }
          if (exe) { shell.openExternal(exe).catch(() => {}); return { success:true, message:`${p.name} खोला।`, data:null }; }
          shell.openExternal(`https://www.google.com/search?q=${encodeURIComponent(name)}`);
          return { success:true, message:`"${name}" सर्च कर रहा हूँ।`, data:null };
        }
        break;

      case "web":

        }

        }
        break;

      default:
        return { success:false, message:`अज्ञात category: ${category}`, data:null };
    }
  } catch (e) {
    return { success:false, message:String(e.message), data:null };
  }
  return { success:false, message:`अज्ञात action: ${category}/${action}`, data:null };
}
