    function resolveBridgeBinary() {
      const exeName = process.platform === "win32"
        ? "pika-bridge.exe"
        : "pika-bridge";

      // Packaged: inside resources/bridge/
      // Dev:      inside python-dist/
      const packaged = path.join(process.resourcesPath, "bridge", exeName);
      const dev = path.join(__dirname, "..", "python-dist", exeName);

      if (app.isPackaged && fs.existsSync(packaged)) return packaged;
      if (fs.existsSync(dev)) return dev;
      return null;   // will fall back to system python in dev

    function startBridge() {
      const bin = resolveBridgeBinary();

      if (bin) {
        bridgeProcess = spawn(bin, [], {
          cwd: path.dirname(bin),
          env: {
            ...process.env,
            PYTHONIOENCODING: "utf-8",
            PYTHONUNBUFFERED: "1",
            PIKA_MODELS_DIR: app.isPackaged
              ? path.join(process.resourcesPath, "models")
              : path.join(ROOT, "models"),

STEP A2.4 — Private NDJSON IPC (replace local WebSocket)

Python must support a `PIKA_IPC_MODE=stdio` mode where it reads
newline-delimited JSON from stdin and writes newline-delimited JSON to stdout.

Request (Electron -> Python), one JSON object per line:

    {"id":"uuid","type":"command","category":"files","action":"create_file","params":{"path":"test.txt"}}

Response (Python -> Electron), one JSON object per line:

    {"id":"uuid","type":"response","status":"success","message":"फाइल बनी","data":null}

Events (Python -> Electron, no id):

    {"type":"event","event":"voice_partial","data":{"text":"chrome kh"}}
    {"type":"event","event":"system_status","data":{"cpu":45,"ram":62}}

Node side (main.cjs) — line buffering is MANDATORY:

    let stdoutBuffer = "";
    bridgeProcess.stdout.on("data", (chunk) => {
      stdoutBuffer += chunk.toString("utf8");
      let idx;
      while ((idx = stdoutBuffer.indexOf("\n")) >= 0) {
        const line = stdoutBuffer.slice(0, idx).trim();
        stdoutBuffer = stdoutBuffer.slice(idx + 1);
        if (!line) continue;
        try { handleBridgeMessage(JSON.parse(line)); }
        catch { console.log("[bridge]", line); }   // plain log line
      }
    });

Pending request map with timeout:

    const pending = new Map();

    function sendToBridge(msg, timeoutMs = 30000) {
      return new Promise((resolve, reject) => {
        if (!bridgeProcess || bridgeProcess.killed) {
          return resolve({ status: "error", message: "Bridge not running" });
        }
        const id = msg.id || randomUUID();
        const timer = setTimeout(() => {
          pending.delete(id);
          resolve({ status: "error", message: "Bridge timeout" });
        }, timeoutMs);

        pending.set(id, { resolve, timer });
        bridgeProcess.stdin.write(JSON.stringify({ ...msg, id }) + "\n");
      });
    }

    function handleBridgeMessage(msg) {
      if (msg.id && pending.has(msg.id)) {
        const { resolve, timer } = pending.get(msg.id);
        clearTimeout(timer);
        pending.delete(msg.id);
        resolve(msg);
        return;
      }
      // Events / llm_stream -> forward to renderer
      if (mainWindow && !mainWindow.isDestroyed()) {
        mainWindow.webContents.send("pika:bridge-event", msg);
      }
    }

Python side must add to pc_bridge.py (keep WebSocket mode too, do NOT remove it):

    async def stdio_loop():
        """NDJSON over stdin/stdout for Electron private IPC."""
        loop = asyncio.get_event_loop()
        reader = asyncio.StreamReader()
        await loop.connect_read_pipe(
            lambda: asyncio.StreamReaderProtocol(reader), sys.stdin
        )

        def emit(obj):
            sys.stdout.write(json.dumps(obj, ensure_ascii=False) + "\n")
            sys.stdout.flush()

        emit({"type": "event", "event": "connection_ready",
              "data": {"server_version": SERVER_VERSION, "mode": "stdio"}})

        asyncio.create_task(status_loop_stdio(emit))

        while True:
            line = await reader.readline()
            if not line:
                break
            try:
                msg = json.loads(line.decode("utf-8").strip())
            except Exception:
                continue
            asyncio.create_task(handle_stdio_message(msg, emit))

    if __name__ == "__main__":
        if os.getenv("PIKA_IPC_MODE") == "stdio":
            asyncio.run(stdio_loop())
        else:
            asyncio.run(main())          # existing WebSocket server

STEP A2.5 — Health monitoring, auto-restart, crash backoff

    let bridgeRestarts = 0;
    const MAX_RESTARTS = 5;

    bridgeProcess.on("close", (code) => {
      bridgeProcess = null;
      mainWindow?.webContents.send("bridge:status", { running: false, code });

      if (!isQuitting && bridgeRestarts < MAX_RESTARTS) {
        bridgeRestarts++;
        const delay = Math.min(1000 * 2 ** bridgeRestarts, 30000);
        setTimeout(startBridge, delay);
      } else if (bridgeRestarts >= MAX_RESTARTS) {
        dialog.showErrorBox(
          "Pika Bridge",
          "Bridge repeatedly crashed. App will run in limited mode."
        );
      }
    });

    // Heartbeat every 15s
    setInterval(async () => {
      const r = await sendToBridge({ type: "ping" }, 5000);
      if (r.status === "error") { stopBridge(); startBridge(); }
    }, 15000);

Reset `bridgeRestarts = 0` on a successful `connection_ready` event.

STEP A2.6 — Guaranteed cleanup (no orphan processes)

    function stopBridge() {
      if (!bridgeProcess) return;
      try {
        if (process.platform === "win32") {
          spawn("taskkill", ["/pid", String(bridgeProcess.pid), "/f", "/t"]);
        } else {
          bridgeProcess.kill("SIGTERM");
          setTimeout(() => bridgeProcess?.kill("SIGKILL"), 3000);
        }
      } catch {}
      bridgeProcess = null;
    }

    app.on("before-quit", () => { isQuitting = true; stopBridge(); });
    app.on("will-quit", () => { globalShortcut.unregisterAll(); stopBridge(); });
    process.on("exit", stopBridge);
    process.on("SIGINT", () => { stopBridge(); process.exit(0); });
