  import sys

  async def stdio_loop():
      """Private IPC for Electron sidecar. Reads NDJSON, writes NDJSON."""
      loop = asyncio.get_event_loop()
      reader = asyncio.StreamReader()
      await loop.connect_read_pipe(
          lambda: asyncio.StreamReaderProtocol(reader),
          sys.stdin.buffer
      )

      def emit(obj: dict) -> None:
          line = json.dumps(obj, ensure_ascii=False) + "\n"
          sys.stdout.buffer.write(line.encode("utf-8"))
          sys.stdout.buffer.flush()

      emit({"type": "event", "event": "connection_ready",
            "data": {"server_version": SERVER_VERSION, "mode": "stdio"},
            "timestamp": datetime.utcnow().isoformat()})

      asyncio.create_task(status_loop_stdio(emit))

      while True:
          raw = await reader.readline()
          if not raw:
              break
          try:
              msg = json.loads(raw.decode("utf-8").strip())
          except Exception:
              continue
          asyncio.create_task(handle_stdio_message(msg, emit))

  async def status_loop_stdio(emit):
      """Push system_status events every 5 seconds over stdout."""
      try:
          while True:
              data = {"cpu": 0, "ram": 0, "battery": None}
              if psutil:
                  data["cpu"] = psutil.cpu_percent()
                  data["ram"] = psutil.virtual_memory().percent
                  b = psutil.sensors_battery()
                  if b:
                      data["battery"] = {"percent": int(b.percent), "plugged": b.power_plugged}
              emit({"type": "event", "event": "system_status", "data": data,
                    "timestamp": datetime.utcnow().isoformat()})
              await asyncio.sleep(5)
      except Exception:
          pass

  async def handle_stdio_message(msg: dict, emit) -> None:
      """Route one NDJSON message and emit the response."""
      try:
          mtype = msg.get("type")
          mid = msg.get("id")

          if mtype == "ping":
              emit({"type": "pong", "id": mid, "timestamp": datetime.utcnow().isoformat()})
              return

          if mtype == "tts_speak":
              params = msg.get("params", {}) or {}
              await emit_event(mid, await generate_tts(
                  params.get("text", ""), params.get("voice", DEFAULT_TTS_VOICE)), emit)
              return

          if mtype == "query":
              # LLM streaming over stdout
              async for chunk, provider, done in llm_stream(
                      ((msg.get("params") or {}).get("text", ""))):
                  emit({"type": "llm_stream", "chunk": chunk, "provider": provider,
                        "id": mid, "done": done,
                        "timestamp": datetime.utcnow().isoformat()})
              return

          if mtype == "command":
              cat = msg.get("category")
              act = msg.get("action")
              if cat == "_confirm":
                  cid = (msg.get("params") or {}).get("confirmation_id")
                  if act == "approve" and cid in PENDING_CONFIRM:
                      original = PENDING_CONFIRM.pop(cid)
                      result = route_command(original)
                  else:
                      PENDING_CONFIRM.pop(cid, None)
                      result = {"success": True, "message": "रद्द किया गया।", "data": None}
              elif (cat, act) in CONFIRM_REQUIRED:
                  cid = str(uuid.uuid4())
                  PENDING_CONFIRM[cid] = msg
                  emit({"type": "response", "status": "confirmation_required",
                        "message": f"क्या आप {cat}/{act} करना चाहते हैं?",
                        "confirmation_id": cid, "id": mid,
                        "timestamp": datetime.utcnow().isoformat()})
                  return
              else:
                  result = route_command(msg)

              emit({"type": "response",
                    "status": "success" if result.get("success") else "error",
                    "message": result.get("message", ""),
                    "data": result.get("data"),
                    "confirmation_id": None,
                    "id": mid,
                    "timestamp": datetime.utcnow().isoformat()})
      except Exception as e:
          emit({"type": "response", "status": "error",
                "message": str(e), "id": msg.get("id"),
                "timestamp": datetime.utcnow().isoformat()})

  # Modify __main__ block:
  if __name__ == "__main__":
      if os.getenv("PIKA_IPC_MODE") == "stdio":
          asyncio.run(stdio_loop())        # Electron sidecar mode
      else:
          asyncio.run(main())              # WebSocket server mode (unchanged)

Add these functions:

  const { randomUUID } = require("crypto");

  let stdoutBuffer = "";
  const pending = new Map();
  let bridgeRestarts = 0;
  const MAX_BRIDGE_RESTARTS = 5;

  function resolveBridgeBinary() {
    const exeName = process.platform === "win32" ? "pika-bridge.exe" : "pika-bridge";
    const packaged = path.join(process.resourcesPath, "bridge", exeName);
    const dev = path.join(__dirname, "..", "python-dist", exeName);
    if (app.isPackaged && fs.existsSync(packaged)) return { bin: packaged, mode: "packaged" };
    if (fs.existsSync(dev)) return { bin: dev, mode: "dev-exe" };
    return null;
  }

  function startBridge() {
    const found = resolveBridgeBinary();
    const bridgeEnv = {
      ...process.env,
      PYTHONIOENCODING: "utf-8",
      PYTHONUNBUFFERED: "1",
      PIKA_IPC_MODE: "stdio",
      PIKA_MODELS_DIR: app.isPackaged
        ? path.join(process.resourcesPath, "models")
        : path.join(ROOT, "models"),
      PIKA_DATA_DIR: app.getPath("userData"),
    };

    if (found) {
      logger.info(`[bridge] Starting sidecar: ${found.bin} (${found.mode})`);
      bridgeProcess = spawn(found.bin, [], {
        cwd: path.dirname(found.bin),
        env: bridgeEnv,
        windowsHide: true,
        stdio: ["pipe", "pipe", "pipe"],
      });
    } else {
      // Dev fallback only — never in production
      const py = process.platform === "win32" ? "python" : "python3";
      const script = path.join(ROOT, "pc_bridge.py");
      logger.warn("[bridge] Sidecar not found, falling back to system Python");
      bridgeProcess = spawn(py, [script], {
        cwd: ROOT,
        env: bridgeEnv,
        windowsHide: true,
        stdio: ["pipe", "pipe", "pipe"],
      });
    }

    // Line-buffered stdout → parse NDJSON
    stdoutBuffer = "";
    bridgeProcess.stdout.on("data", (chunk) => {
      stdoutBuffer += chunk.toString("utf8");
      let idx;
      while ((idx = stdoutBuffer.indexOf("\n")) >= 0) {
        const line = stdoutBuffer.slice(0, idx).trim();
        stdoutBuffer = stdoutBuffer.slice(idx + 1);
        if (!line) continue;
        try { handleBridgeMessage(JSON.parse(line)); }
        catch { logger.info(`[bridge] ${line}`); }
      }
    });

    bridgeProcess.stderr.on("data", (d) => logger.error(`[bridge:err] ${d.toString().trim()}`));

    bridgeProcess.on("close", (code) => {
      bridgeProcess = null;
      pending.forEach(({ resolve, timer }) => {
        clearTimeout(timer);
        resolve({ type: "response", status: "error", message: "Bridge stopped", data: null });
      });
      pending.clear();

      if (mainWindow && !mainWindow.isDestroyed()) {
        mainWindow.webContents.send("bridge:status", { running: false, code });
      }

      if (!isQuitting && bridgeRestarts < MAX_BRIDGE_RESTARTS) {
        bridgeRestarts++;
        const delay = Math.min(1000 * Math.pow(2, bridgeRestarts), 30000);
        logger.info(`[bridge] Restart ${bridgeRestarts}/${MAX_BRIDGE_RESTARTS} in ${delay}ms`);
        setTimeout(startBridge, delay);
      } else if (bridgeRestarts >= MAX_BRIDGE_RESTARTS) {
        dialog.showErrorBox("Pika Bridge",
          "Bridge crashed too many times. Running in limited demo mode.\nPlease restart the app.");
      }
    });

    bridgeProcess.on("error", (err) => logger.error(`[bridge] Spawn error: ${err.message}`));
  }

  function handleBridgeMessage(msg) {
    // Forward system events / llm_stream to renderer
    if (mainWindow && !mainWindow.isDestroyed()) {
      mainWindow.webContents.send("pika:bridge-event", msg);
    }
    // Resolve pending request if this is a response/pong
    if (msg.id && pending.has(msg.id)) {
      const { resolve, timer } = pending.get(msg.id);
      clearTimeout(timer);
      pending.delete(msg.id);
      resolve(msg);
    }
    // Reset restart counter on successful connection
    if (msg.event === "connection_ready") {
      bridgeRestarts = 0;
      if (mainWindow && !mainWindow.isDestroyed()) {
        mainWindow.webContents.send("bridge:status", { running: true, mode: msg.data?.mode });
      }
    }
  }

  function sendToBridge(msg, timeoutMs = 30000) {
    return new Promise((resolve) => {
      if (!bridgeProcess || bridgeProcess.killed) {
        return resolve({ type: "response", status: "error",
                        message: "Bridge not running", data: null });
      }
      const id = msg.id || randomUUID();
      const timer = setTimeout(() => {
        pending.delete(id);
        resolve({ type: "response", status: "error", message: "Bridge timeout", data: null });
      }, timeoutMs);
      pending.set(id, { resolve, timer });
      try {
        bridgeProcess.stdin.write(JSON.stringify({ ...msg, id }) + "\n");
      } catch (err) {
        clearTimeout(timer);
        pending.delete(id);
        resolve({ type: "response", status: "error", message: err.message, data: null });
      }
    });
  }

  function stopBridge() {
    if (!bridgeProcess) return;
    try {
      if (process.platform === "win32") {
        spawn("taskkill", ["/pid", String(bridgeProcess.pid), "/f", "/t"],
              { windowsHide: true });
      } else {
        bridgeProcess.kill("SIGTERM");
        setTimeout(() => bridgeProcess?.kill("SIGKILL"), 3000);
      }
    } catch {}
    bridgeProcess = null;
  }

  // Heartbeat every 15 seconds
  let heartbeatTimer = null;
  function startHeartbeat() {
    heartbeatTimer = setInterval(async () => {
      if (!bridgeProcess) return;
      const r = await sendToBridge({ type: "ping" }, 5000);
      if (r.status === "error") {
        logger.warn("[bridge] Heartbeat failed, restarting...");
        stopBridge();
        startBridge();
      }
    }, 15000);
  }

=== STEP 7: Add mobile gateway in electron/services/mobileServer.cjs ===

This is a small Express server that:
- Serves dist/ on http://0.0.0.0:3000 (phone browser UI)
- Provides WebSocket for phone commands
- Requires a 6-digit PIN before accepting commands
- Forwards authenticated commands to Electron main via a callback
- Destructive commands trigger a desktop confirmation dialog

'use strict';
const express = require("express");
const http = require("http");
const { WebSocketServer } = require("ws");
const path = require("path");
const crypto = require("crypto");
const os = require("os");

const MOBILE_PORT = 3000;
const TOKEN_VALIDITY_MS = 24 * 60 * 60 * 1000; // 24h

function generatePin() {
  return String(Math.floor(100000 + Math.random() * 900000));
}

function getLanIp() {
  for (const iface of Object.values(os.networkInterfaces())) {
    for (const info of (iface || [])) {
      if (info.family === "IPv4" && !info.internal) return info.address;
    }
  }
  return "127.0.0.1";
}

class MobileGateway {
  constructor({ distDir, onCommand, onShowPin }) {
    this.distDir = distDir;
    this.onCommand = onCommand;   // (msg) => Promise<result>
    this.onShowPin = onShowPin;   // (pin, lanIp) => void
    this.currentPin = generatePin();
    this.tokens = new Map();      // token -> expires
    this.app = null;
    this.server = null;
    this.wss = null;
    this.clients = new Set();
  }

  _generateToken(pin) {
    if (pin !== this.currentPin) return null;
    const token = crypto.randomBytes(32).toString("hex");
    this.tokens.set(token, Date.now() + TOKEN_VALIDITY_MS);
    this.currentPin = generatePin();  // rotate PIN
    return token;
  }

  _validateToken(token) {
    const exp = this.tokens.get(token);
    if (!exp) return false;
    if (Date.now() > exp) { this.tokens.delete(token); return false; }
    return true;
  }

  start() {
    this.app = express();
    this.app.use(express.json());

    // Serve React build
    this.app.use(express.static(this.distDir));

    // Pairing endpoint
    this.app.post("/api/pair", (req, res) => {
      const pin = String(req.body?.pin || "").trim();
      const token = this._generateToken(pin);
      if (!token) {
        return res.status(401).json({ error: "Invalid PIN" });
      }
      this.onShowPin(null, getLanIp());  // notify main: pin was used
      res.json({ token, expires: Date.now() + TOKEN_VALIDITY_MS });
    });

    // Show current PIN (GET — main process calls this to get PIN for QR)
    this.app.get("/api/pair-info", (req, res) => {
      res.json({ pin: this.currentPin, ip: getLanIp(), port: MOBILE_PORT });
    });

    // SPA fallback
    this.app.get("*", (req, res) => {
      res.sendFile(path.join(this.distDir, "index.html"));
    });

    this.server = http.createServer(this.app);
    this.wss = new WebSocketServer({ server: this.server });

    this.wss.on("connection", (ws) => {
      let authenticated = false;
      this.clients.add(ws);

      ws.on("message", async (raw) => {
        let msg;
        try { msg = JSON.parse(raw.toString()); } catch { return; }

        if (!authenticated) {
          if (msg.type === "auth" && this._validateToken(msg.token)) {
            authenticated = true;
            ws.send(JSON.stringify({ type: "auth_ok" }));
          } else {
            ws.send(JSON.stringify({ type: "auth_fail" }));
          }
          return;
        }

        const result = await this.onCommand(msg);
        ws.send(JSON.stringify({ ...result, id: msg.id }));
      });

      ws.on("close", () => this.clients.delete(ws));
    });

    this.server.listen(MOBILE_PORT, "0.0.0.0", () => {
      const ip = getLanIp();
      this.onShowPin(this.currentPin, ip);
      console.log(`[mobile] Gateway on http://${ip}:${MOBILE_PORT}`);
    });
  }

  broadcast(msg) {
    const str = JSON.stringify(msg);
    for (const ws of this.clients) {
      if (ws.readyState === 1) ws.send(str);
    }
  }

  stop() {
    this.wss?.close();
    this.server?.close();
  }
}
